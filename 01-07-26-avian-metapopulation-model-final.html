Modifications

This version has balanced population dynamics, standardised temperature using briere function and includes the bird breeding season. 1 infected bird is imported into each population in March of each year. Simulation runs over 5 years.

Equations

\[ \frac{\mathrm{d}S_i}{\mathrm{d}t} = (\gamma N_i) - (\lambda S_i) - (\mu S_i) \] \[ \frac{\mathrm{d}E_i}{\mathrm{d}t} = (\lambda S_i) - (\theta E_i) - (\mu E_i) \] \[ \frac{\mathrm{d}I_i}{\mathrm{d}t} = (\theta E_i) - (\mu I_i) - (\sigma I_i) + seed \] \[ \frac{\mathrm{d}R_i}{\mathrm{d}t} = (\sigma I_i) - (\mu R_i) \] \[ \lambda = briere(beta_{i,1} \frac {I_1}{N_1} + beta_{i,2} \frac {I_2}{N_2} + beta_{i,3} \frac {I_3}{N_3}+ beta_{i,4} \frac {I_4}{N_4}+ beta_{i,5} \frac {I_5}{N_5}) \]

Matrix

\[ \begin{aligned} beta &= \begin{bmatrix} 3.0 & 0.01 & 0.01 \\ 0.01 & 3.0 & 0.01 \\ 0.01 & 0.01 & 3.0 \end{bmatrix} \qquad beta &= \begin{bmatrix} \beta_{11} & \beta_{12} & \beta_{13}\\ \beta_{21} & \beta_{22} & \beta_{23}\\ \beta_{31} & \beta_{32} & \beta_{33} \end{bmatrix} \end{aligned} \]

low connectivity is represented by beta = 0.01 moderate connectivity is represented by beta = 0.05 high connectivity is represented by beta = 0.1

Random seed

A random number is chosen 1-5 (10 times) to determine which population receives the infection seed

SIR model

sir <- odin({
  
  # Equations for transitions between compartments by group
  
  deriv(S[]) <- (gamma*N[i]) - (lambda[i]*S[i]) - (mu*S[i]) - seed[i]
  
  deriv(E[]) <- (lambda[i]*S[i]) - (theta*E[i]) - (mu*E[i])
  
  deriv(I[]) <- (theta*E[i]) - (mu*I[i]) - (sigma*I[i]) + seed[i]
  
  deriv (R[]) <- (sigma*I[i]) - (mu*R[i])
  
  # with more populations it became easier to separate lambda
  
  lambda[] <- briere * (
    beta[i,1] * I[1]/N[1] +
      beta[i,2] * I[2]/N[2] +
      beta[i,3] * I[3]/N[3] +
      beta[i,4] * I[4]/N[4] +
      beta[i,5] * I[5]/N[5]
  )
  
  beta <- parameter()
  
  
  # initial conditions
  
  N[] <- S[i] + E[i] + I[i] + R[i]
  
  initial(S[]) <- 600 
  
  initial(E[]) <- 0
  
  initial(I[]) <- 0 
  
  initial(R[]) <- 0
  
  # infection seeded on day 60 each year
  year <- floor(time/365) + 1
  seed[] <- if(
    i == seed_pop[year] &&
    time%%365 >= 60 &&
    time%%365 < 61
  ) I0 else 0
  
  # User defined parameters:
  
  briere <- interpolate(briere_time, briere_value, "linear")
  
  briere_time <- parameter(constant = TRUE)
  briere_value <- parameter(constant = TRUE)
  
  seed_pop <- parameter(constant = TRUE)
  
  I0 <- parameter(10) # seed is 10 infected birds
  
  mu <- parameter(0.0015) # sensitivity analysis 0.00382
  
  theta <- parameter(0.5)
  
  sigma <- parameter(0.57) # house sparrow
  # sensitivity analysis: 0.2
  
  # birth season may-july
  gamma <- if (time %% 365 >= 121 && time %% 365 <= 213) 0.0059 else 0 #0.5 births per bird
  # sensitivity analysis  0.015 (1.5 births per bird)
  
  # Dimensions of arrays
  
  dim(S) <- 5
  
  dim(E) <- 5
  
  dim(I) <- 5
  
  dim(R) <- 5
  
  dim(N) <- 5
  
  dim(beta) <- c(5, 5)
  
  dim(lambda) <- 5
  
  dim(briere_time, briere_value) <- parameter(rank = 1)
  
  dim(seed_pop) <- 10
  
  dim(seed) <- 10
})

BriereNorm <- function(temp, T_minB, T_maxB, mB) {
  
  briere <- (temp - T_minB) * (T_maxB - temp)^mB
  
  briere[briere < 0] <- 0
  briere[is.na(briere)] <- 0
  
  briere <- briere / max(briere, na.rm = TRUE)
  
  return(briere)
}

sys <- dust_system_create(
  sir,
  list(
    beta = beta,
    briere_time = briere_time,
    briere_value = briere_value,
    seed_pop=seed_pop
  )
)

dust_system_set_state_initial(sys)

t <- seq(0, 3649) # 5 years

y <- dust_system_simulate(sys, t)

y <- dust_unpack_state(sys, y)
