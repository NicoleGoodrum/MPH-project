Modifications

This version has balanced population dynamics, standardised temperature using briere function and includes the bird breeding season. 1 infected bird is imported into each population in March of each year. Simulation runs over 5 years.

Equations

\[ \frac{\mathrm{d}S_i}{\mathrm{d}t} = (\gamma N_i) - (briere(m_{i,1} \frac {I_1}{N_1} + m_{i,2} \frac {I_2}{N_2} + m_{i,3} \frac {I_3}{N_3})S_i) - (\mu S_i) \] \[ \frac{\mathrm{d}E_i}{\mathrm{d}t} = (briere(m_{i,1} \frac {I_1}{N_1} + m_{i,2} \frac {I_2}{N_2} + m_{i,3} \frac {I_3}{N_3})S_i) - (\theta E_i) - (\mu E_i) \] \[ \frac{\mathrm{d}I_i}{\mathrm{d}t} = (\theta E_i) - (\mu I_i) - (\sigma I_i) + seed \] \[ \frac{\mathrm{d}R_i}{\mathrm{d}t} = (\sigma I_i) - (\mu R_i) \]

Matrix

\[ \begin{aligned} m &= \begin{bmatrix} 0.9 & 0.15 & 0.02 \\ 0.15 & 0.55 & 0.08 \\ 0.02 & 0.08 & 0.25 \end{bmatrix} \qquad m &= \begin{bmatrix} \beta_{11} & \beta_{12} & \beta_{13}\\ \beta_{21} & \beta_{22} & \beta_{23}\\ \beta_{31} & \beta_{32} & \beta_{33} \end{bmatrix} \end{aligned} \] Population 1 has high competence, P2 intermediate and P3 low P1 and P2 are nearby, P2 and P3 are moderately separated and P3 and P1 are far apart

SIR model

sir <- odin({
  
  # Equations for transitions between compartments by group
  
  deriv(S[]) <- (gamma*N[i]) - (briere*(m[i,1] * I[1]/N[1] + m[i,2] * I[2]/N[2] + m[i,3] * I[3]/N[3])*S[i]) - (mu*S[i]) - seed
  
  deriv(E[]) <- (briere*(m[i,1] * I[1]/N[1] + m[i,2] * I[2]/N[2] + m[i,3] * I[3]/N[3])*S[i]) - (theta*E[i]) - (mu*E[i])
  
  deriv(I[]) <- (theta*E[i]) - (mu*I[i]) - (sigma*I[i]) + seed #seed imports 1 infection every march
  
  deriv (R[]) <- (sigma*I[i]) - (mu*R[i])
  
  m <- parameter()
  
  
  # initial conditions
  
  N[] <- S[i] + E[i] + I[i] + R[i]
  
  initial(S[]) <- 500 - I0
  
  initial(E[]) <- 0
  
  initial(I[]) <- 0 
  
  initial(R[]) <- 0
  
  
  seed <- if(time%%365>=60 && time%%365<61)I0 else 0 #if march 1st I0 = 1
  
  # User defined parameters:
  
  briere <- interpolate(briere_time, briere_value, "linear") # temperature function
  briere_time <- parameter(constant = TRUE)
  briere_value <- parameter(constant = TRUE)
  
  I0 <- parameter(1)
  
  mu <- parameter(0.0015) 
  
  theta <- parameter(0.5)
  
  sigma <- parameter(0.57)
  
  #gamma <- parameter(0.0015) # tested with constant gamma
  gamma <- if (time %% 365 >= 121 && time %% 365 <= 213) 0.0059 else 0 #(mu*365)/93 #may june july (93 days)
  
  # Dimensions of arrays
  
  dim(S) <- 3
  
  dim(E) <- 3
  
  dim(I) <- 3
  
  dim(R) <- 3
  
  dim(N) <- 3
  
  dim(m) <- c(3, 3)
  
  dim(briere_time, briere_value) <- parameter(rank = 1)
})

daily_temperatures_MO <- read.csv("~/daily_temperatures.csv") # imported dataset (.csv) and only included march-october 
daily_temp <- daily_temperatures_MO$DAILY_AVG 

BriereNorm <- function(temp, T_minB, T_maxB, mB) {
  
  briere <- (temp - T_minB) * (T_maxB - temp)^mB
  
  briere[briere < 0] <- 0
  briere[is.na(briere)] <- 0
  
  briere <- briere / max(briere, na.rm = TRUE)
  
  return(briere)
}


briere_value <- rep(BriereNorm(
  temp = daily_temp, 
  T_minB = 0.3,
  T_maxB = 31.15,
  mB = 0.3), # peak at 24 degrees
  5
)

briere_time <- seq(0, length(briere_value)-1, by=1)

temp_seq <- seq(0.3, 31.15, by = 0.1)

plot(temp_seq,
     BriereNorm(temp_seq, 0.3, 31.15, 0.5),
     type = "l",
     xlab = "Temperature (°C)",
     ylab = "Brière response")

  list(
    m = m,
    briere_time = briere_time,
    briere_value = briere_value
  )
)

t <- seq(0, 1824) # 5 years

y <- dust_system_simulate(sys, t)

y <- dust_unpack_state(sys, y)
